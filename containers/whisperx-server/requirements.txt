# whisperx-server container deps — pinned to ai-transcriber's known-good
# working set (CUDA 11.8 / Pascal-compatible). Rationale lives in
# ai-transcriber's requirements.txt and gpu-model-broker.md Decisions log;
# the short version:
#   - torch 2.4.0+cu118: GTX 1070 needs CUDA 11.8 wheels; torch 2.8+ is
#     CUDA 12.x only and drops Pascal compute capability 6.1
#   - whisperx<3.8: 3.8+ requires torch>=2.8
#   - pyannote-audio<4.0: 4.0+ requires torch>=2.8
#   - nvidia-cudnn-cu12==9.1.0.70: 9.20+ trips CUDNN_STATUS_EXECUTION_FAILED
#     on Pascal
#
# This whole pin set unwinds when the GPU upgrade lands (5060 Ti, 3-6 month
# horizon per self-hosted-brain-sovereignty). Keep it frozen until then.

--extra-index-url https://download.pytorch.org/whl/cu118

torch==2.4.0+cu118
torchvision==0.19.0+cu118
torchaudio==2.4.0+cu118

whisperx==3.3.2
pyannote-audio==3.3.2
faster-whisper==1.1.0

# Phase 2.5 (2026-06-13): pyannote-audio 3.3.2 imports matplotlib at module
# init (pyannote.audio.tasks.segmentation.mixins) but does NOT declare it as a
# hard install requirement — wheel resolver skips it. Pin explicitly so the
# baked image works offline. matplotlib 3.10 series is the current stable;
# pinned narrowly to avoid surprises.
matplotlib>=3.7,<4.0

# Phase 2.5 (2026-06-13): pyannote-audio 3.3.2's Pipeline.from_pretrained()
# still calls hf_hub_download(use_auth_token=...) — that kwarg was removed
# from huggingface_hub in 0.30.0 (Oct 2024) in favor of token=. Cap hf-hub
# to a co-tested pyannote-3.3.2-era version. Lifts when we move past
# pyannote-audio 4.0+ (which requires torch>=2.8 and is blocked by Pascal).
huggingface-hub>=0.20,<0.27

nvidia-cudnn-cu12==9.1.0.70

fastapi==0.115.6
uvicorn[standard]==0.34.0
pydantic==2.10.4
